    start: int,

    in_string = False
    escape = False
    gemma_quote = '<|"|>'

    i = start

    while i < len(text):
        if text.startswith(gemma_quote, i):
            in_string = not in_string
            i += len(gemma_quote)
            continue

        ch = text[i]

        if escape:
            escape = False
            i += 1
            continue

        if ch == "\\" and in_string:
            escape = True
            i += 1
            continue

        if ch == '"':
            in_string = not in_string
            i += 1

        if not in_string:

                depth += 1

def split_top_level(
    text: str,
    separator: str = ",",
) -> List[str]:
    pieces = []
    current = []

    brace_depth = 0
    bracket_depth = 0
    paren_depth = 0

    in_string = False
    escape = False
    gemma_quote = '<|"|>'

    i = 0

    while i < len(text):
        if text.startswith(gemma_quote, i):
            in_string = not in_string
            current.append(gemma_quote)
            i += len(gemma_quote)
            continue

        ch = text[i]

        if escape:
            current.append(ch)
            escape = False
            i += 1
            continue

        if ch == "\\" and in_string:
            current.append(ch)
            escape = True
            i += 1
            continue

        if ch == '"':
            in_string = not in_string
            current.append(ch)
            i += 1
            continue

        if not in_string:
            if ch == "{":
                brace_depth += 1
            elif ch == "}":
                brace_depth -= 1
            elif ch == "[":
                bracket_depth += 1
            elif ch == "]":
                bracket_depth -= 1
            elif ch == "(":
                paren_depth += 1
            elif ch == ")":
                paren_depth -= 1

            elif (

                and bracket_depth == 0
                and paren_depth == 0

                    pieces.append(piece)

                current = []
                i += 1
                continue

        current.append(ch)
        i += 1

    if final_piece:

    return pieces


def split_key_value(text: str) -> Optional[Tuple[str, str]]:
    brace_depth = 0
    bracket_depth = 0

    in_string = False
    escape = False
    gemma_quote = '<|"|>'

    i = 0

    while i < len(text):
        if text.startswith(gemma_quote, i):
            in_string = not in_string
            i += len(gemma_quote)
            continue

        ch = text[i]

            i += 1

            i += 1
            continue

        if ch == '"':
            in_string = not in_string
            i += 1
            continue

                bracket_depth += 1
            elif ch == "]":

def parse_gemma_value(raw: str) -> Any:
    raw = raw.strip()

    if (
        raw.startswith('<|"|>')
        and raw.endswith('<|"|>')
        and len(raw) >= 10
    ):
        value = raw[len('<|"|>'):-len('<|"|>')]

        return (
            value
            .replace('\\"', '"')
            .replace("\\\\", "\\")
        )

    try:
        return json.loads(raw)
    except Exception:
        pass

    lower = raw.lower()

    if lower == "true":
        return True

    if lower == "false":
        return False

    if lower in ("null", "none"):
        return None

    try:
        if "." in raw:
            return float(raw)

        return int(raw)

    except Exception:
        pass

    return raw.strip('"')


def parse_gemma_arguments(raw_args: str) -> Dict[str, Any]:
    raw_args = raw_args.strip()

    if not raw_args:
        return {}

    cleaned = raw_args.replace('<|"|>', '"')

    try:
        parsed = json.loads("{" + cleaned + "}")

        if isinstance(parsed, dict):
            return parsed

    except Exception:
        pass

    result: Dict[str, Any] = {}

    for piece in split_top_level(raw_args):
        pair = split_key_value(piece)

        if pair is None:
            continue

        key, raw_value = pair
        key = key.strip().strip('"')

        result[key] = parse_gemma_value(raw_value)

    if result:
        return result

    return {"input": raw_args}


def parse_gemma_tool_calls(
    content: Any,
) -> Optional[List[Dict[str, Any]]]:
    if not isinstance(content, str):
        return None

    marker = "<|tool_call>"

    if marker not in content:
        return None

    calls: List[Dict[str, Any]] = []
    position = 0

    while True:
        marker_position = content.find(marker, position)

        if marker_position < 0:
            break

        cursor = marker_position + len(marker)

        while cursor < len(content) and content[cursor].isspace():
            cursor += 1

        if content.startswith("call:", cursor):
            cursor += len("call:")

        name_match = re.match(
            r"[A-Za-z0-9_.\-]+",
            content[cursor:],
        )

        if not name_match:
            position = cursor + 1
            continue

        name = name_match.group(0)
        cursor += len(name)

        while cursor < len(content) and content[cursor].isspace():
            cursor += 1

        if cursor >= len(content) or content[cursor] != "{":
            position = cursor + 1
            continue

        try:
            raw_args, after_brace = extract_balanced_braces(
                content,
                cursor,
            )

        except Exception as exc:
            print(
                f"[tool-parser] failed extracting {name}: {exc}",
                flush=True,
            )

            position = cursor + 1
            continue

        try:
            arguments = parse_gemma_arguments(raw_args)

        except Exception as exc:
            print(
                f"[tool-parser] failed parsing {name}: {exc}",
                flush=True,
            )

            arguments = {"input": raw_args}

        calls.append(
            {
                "id": "call_" + uuid.uuid4().hex[:24],
                "type": "function",
                "function": {
                    "name": name,
                    "arguments": json.dumps(
                        arguments,
                        ensure_ascii=False,
                    ),
                },
            }
        )

        end_marker = content.find(
            "<tool_call|>",
            after_brace,
        )

        if end_marker >= 0:
            position = end_marker + len("<tool_call|>")
        else:
            position = after_brace

    return calls or None


def normalize_completion(

    choice = result["choices"][0]
    message = choice.get("message") or {}

    content = message.get("content")

    if not tool_calls and content:
        try:
            tool_calls = parse_gemma_tool_calls(content)
        except Exception:
            traceback.print_exc()
            tool_calls = None

    normalized_message: Dict[str, Any] = {
        "role": "assistant",
        "content": content,
    }

    if tool_calls:
        normalized_calls = []

        for tc in tool_calls:
            tc = dict(tc)

            if not tc.get("id"):
                tc["id"] = "call_" + uuid.uuid4().hex[:24]

            tc["type"] = "function"

            function = dict(tc.get("function") or {})
            arguments = function.get("arguments", "{}")

            if not isinstance(arguments, str):
                arguments = json.dumps(
                    arguments,
                    ensure_ascii=False,
                )

            function["arguments"] = arguments
            tc["function"] = function

            normalized_calls.append(tc)

        normalized_message["content"] = None
        normalized_message["tool_calls"] = normalized_calls

        finish_reason = "tool_calls"

    else:
        finish_reason = choice.get("finish_reason") or "stop"

    return normalized_message, finish_reason


def sse_line(data: Any) -> str:
    if data == "[DONE]":
        return "data: [DONE]\n\n"

    return (
        "data: "

        + "\n\n"
    )
